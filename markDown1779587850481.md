def fetch_target_ips():

    """Busca IPs ativos mapeados em cabos submarinos e pontos de ancoragem."""

    query = """

        SELECT DISTINCT e.ip_address, e.cable_id, c.name, lp.country_code

        FROM endpoint_ip_samples e

        LEFT JOIN cables c ON e.cable_id = c.id

        LEFT JOIN landing_points lp ON c.id = lp.cable_id

        WHERE e.is_active = TRUE AND e.ip_address IS NOT NULL;

    """

    targets = []

    try:

        with get_connection() as conn:

            with conn.cursor(cursor_factory=psycopg2.extras.DictCursor) as cur:

                cur.execute(query)

                targets = [dict(row) for row in cur.fetchall()]

    except Exception as e:

        log.error(f"Erro ao buscar IPs alvos do banco: {e}")

    return targets

def probe_bgp_prefix(target):

    """Consulta visibilidade BGP e ROA para o prefixo/IP via RIPE STAT."""

    ip = target['ip_address']

    # Converte IP para prefixo de rede genérico para consulta BGP se necessário

    try:

        net = ipaddress.ip_network(ip, strict=False)

        prefix = str(net)

    except Exception:

        prefix = ip



    url = f"{RIPE_STAT_BASE}/routing-status/data.json?resource={ip}"

    try:

        res = requests.get(url, timeout=REQUEST_TIMEOUT)

        if res.status_code == 200:

            data = res.json().get('data', {})

            # Tratamento simplificado dos dados de visibilidade

            visibility_pct = data.get('visibility_percentage', 100.0)

            bgp_state = data.get('bgp_state', [])

            

            peers_seeing = len(bgp_state)

            origin_asn = bgp_state[0].get('origin_as', 'Unknown') if bgp_state else 'Unknown'

            as_path = json.dumps([p.get('as_path', []) for p in bgp_state])



            # Relação com tabela nova

            with get_connection() as conn:

                with conn.cursor() as cur:

                    cur.execute("""

                        INSERT INTO bgp_probe_results 

                        (prefix, cable_id, cable_name, landing_country, peers_seeing, origin_asn, as_path, visibility_pct, raw_json)

                        VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s)

                    """, (prefix, target.get('cable_id'), target.get('name'), target.get('country_code'), 

                          peers_seeing, origin_asn, as_path, visibility_pct, json.dumps(data)))

                    conn.commit()

            log.info(f"BGP Probe Sucesso para o IP/Prefixo: {prefix}")

    except Exception as e:

        log.error(f"Falha na sonda BGP para {ip}: {e}")

def fetch_snmp_value(target_ip, oid):

    """Executa um SNMP GetCmd síncrono para um OID específico."""

    if not SNMP_AVAILABLE:

        return None, "Biblioteca pysnmp não instalada."

    try:

        error_indication, error_status, error_index, var_binds = next(

            getCmd(SnmpEngine(),

                   CommunityData(SNMP_COMMUNITY, mpModel=1),

                   UdpTransportTarget((target_ip, 161), timeout=2, retries=1),

                   ContextData(),

                   ObjectType(ObjectIdentity(oid)))

        )

        if error_indication:

            return None, str(error_indication)

        elif error_status:

            return None, str(error_status)

        else:

            return str(var_binds[0][1]), None

    except Exception as e:

        return None, str(e)



def probe_snmp_target(target):

    """Sonda todas as MIBs parametrizadas para o IP alvo."""

    ip = target['ip_address']

    results = {'target_ip': ip, 'cable_id': target.get('cable_id')}

    error_msg = None



    for key, oid in SNMP_OIDS.items():

        val, err = fetch_snmp_value(ip, oid)

        if err:

            error_msg = err

            results[key] = None

        else:

            results[key] = val



    # Higienização e tipagem para inserção SQL

    try:

        with get_connection() as conn:

            with conn.cursor() as cur:

                cur.execute("""

                    INSERT INTO snmp_probe_results 

                    (target_ip, cable_id, sys_descr, sys_uptime_ticks, if_oper_status, in_octets, out_octets, error_message)

                    VALUES (%s, %s, %s, %s, %s, %s, %s, %s)

                """, (

                    ip, results['cable_id'], results.get('sysDescr'),

                    int(results['sysUpTime']) if results.get('sysUpTime') else None,

                    results.get('ifOperStatus'),

                    int(results['ifInOctets']) if results.get('ifInOctets') else None,

                    int(results['ifOutOctets']) if results.get('ifOutOctets') else None,

                    error_msg

                ))

                conn.commit()

        log.info(f"SNMP Probe finalizada para o host: {ip}")

    except Exception as e:

        log.error(f"Erro ao salvar telemetria SNMP para {ip}: {e}")

# MONITORAÇÃO DE REDE PASSIVA (SCAPY: LLDP / NETFLOW)

# ═══════════════════════════════════════════════════════════════════════════════

def packet_callback(packet):

    """Callback do Scapy para processar pacotes em tempo real e salvar no banco."""

    try:

        # 1. Processamento LLDP (Camada 2 Link-Layer Topology)

        if packet.haslayer('Link Layer Discovery Protocol'):

            lldp = packet['Link Layer Discovery Protocol']

            # Extração de campos básicos do formato TLV do Scapy LLDP

            local_port = packet.summary()

            with get_connection() as conn:

                with conn.cursor() as cur:

                    cur.execute("""

                        INSERT INTO lldp_topology_map (local_device, local_port, neighbor_device)

                        VALUES (%s, %s, %s)

                    """, ("Probe_Local_Host", local_port, "Discovered_Neighbor"))

                    conn.commit()

            log.info("Topologia de infraestrutura atualizada via frame LLDP capturado.")



        # 2. Processamento simplificado NetFlow v5 (Porta Comum 2055)

        elif packet.haslayer('UDP') and packet['UDP'].dport == 2055:

            raw_payload = packet['UDP'].payload.load

            # Decodificação básica do cabeçalho Netflow v5 (primeiros 16 bytes)

            version, count = struct.unpack('!HH', raw_payload[:4])

            if version == 5:

                # Loop através dos registros de fluxo contidos no pacote Netflow

                offset = 24  # Onde começam os records v5

                for _ in range(min(count, 30)):

                    if len(raw_payload) < offset + 48:

                        break

                    flow_data = struct.unpack('!IIIIIIJ', raw_payload[offset:offset+28])

                    src_ip = socket.inet_ntoa(struct.pack('!I', flow_data[0]))

                    dst_ip = socket.inet_ntoa(struct.pack('!I', flow_data[1]))

                    

                    with get_connection() as conn:

                        with conn.cursor() as cur:

                            cur.execute("""

                                INSERT INTO netflow_traffic_samples (src_ip, dst_ip, packets, bytes)

                                VALUES (%s, %s, %s, %s)

                            """, (src_ip, dst_ip, flow_data[4], flow_data[5]))

                            conn.commit()

                    offset += 48

    except Exception as e:

        # Silencia erros de parsing interno para não quebrar a captura em tempo real

        pass



def start_passive_sniffing(interface):

    """Inicia thread de escuta passiva na interface de rede."""

    if not SCAPY_AVAILABLE:

        log.error("Scapy não está instalado. Modo de captura passiva indisponível.")

        return

    log.info(f"Iniciando captura passiva na interface {interface} (Procurando LLDP/NetFlow)...")

    scapy.sniff(iface=interface, prn=packet_callback, store=0)
